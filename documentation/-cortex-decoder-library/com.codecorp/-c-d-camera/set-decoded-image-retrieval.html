setDecodedImageRetrieval

public void setDecodedImageRetrieval(boolean enable)

setDecodedImageRetrieval provides a way to enable retrieving image of a successfully decoded barcode as part of CDResult. By default, this setting is disabled and image will be null unless enabled. 

Sample usage
// Enable
CDCamera.shared.setDecodedImageRetrieval(true);
// Disable
CDCamera.shared.setDecodedImageRetrieval(false);

// Following method can be leveraged to store {CDResult#image} as Bitmap
private Bitmap createBitmapFromRaw(
   ByteBuffer pixBuf, // ByteBuffer from the {CDResult#image}
   int width, // width from the {CDCamera#getResolution()}
   int height // height from the {CDCamera#getResolution()}
  ) {
     pixBuf.rewind();
     int size = width * height;
     int[] rgbData = new int[size];
     for (int i = 0; i < width * height; i++) {
         int y = pixBuf.get() & 0xff;
         rgbData[i] = 0xFF000000 | (y << 16) | (y << 8) | y;
     }
     Bitmap grayImage = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888);
     grayImage.setPixels(rgbData, 0, width, 0, 0, width, height);
     return grayImage;
 }
