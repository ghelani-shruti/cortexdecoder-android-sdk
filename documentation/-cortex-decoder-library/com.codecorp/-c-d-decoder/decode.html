decode

public void decode(Array<byte> imageData, int width, int height, int stride, @NonNull() CDImageDecodeListener decodeListener)

decode scans the image and decodes the barcodes present in the image. 

Sample usage
CDDecoder.shared.decode(byteArray, width, height, stride, this:: onDecode);


 // onDecode callback listener
 void onDecode(CDResult[] cdResults){
    // cdResults is array of decoded items
    if(cdResults[0].status == CDResult.CDDecodeStatus.success){
        // set decoding to true to do another scan
        CDDecoder.shared.setDecoding(true);
    }
 }
