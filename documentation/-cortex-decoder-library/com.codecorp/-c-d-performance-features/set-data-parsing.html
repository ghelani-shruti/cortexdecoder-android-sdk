setDataParsing

public void setDataParsing(CDPerformanceFeatures.CDDataParsing parsingType, String configStr)

setDataParsing is used to select a Data Parsing option. By default, Data Parsing is disable. 

Sample usage
//disable
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.disable,"");
//dlParsing
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.dlParsing,"00000308130328130331091845584578130648320668320678320681010");
//strMatchReplace
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.strMatchReplace,"00+_aI+_aI!,,|12^0OK:^B019\\x01@");
//gs1Parsing
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.gs1Parsing,"00+_aI+_aI|;^3^C#/0D016\\x01@");
//isoParsing
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.isoParsing,"00+_aI+_aI00C\\x01@");
//udiParsing
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.udiParsing,"00+_aI+_aI00C\\x01@");
//jsonParsing
CDPerformanceFeatures.shared.setDataParsing(CDPerformanceFeatures.CDDataParsing.jsonParsing,"00000308130328130331091845584578130648320668320678320681010");
